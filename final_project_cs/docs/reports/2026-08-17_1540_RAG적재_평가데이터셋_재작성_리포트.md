  ```powershell
  python -m eval.runners.proposed  --dataset eval/datasets/golden.jsonl  --repeats 3 --seed 7 --provider openai
  python -m eval.runners.baseline_a --dataset eval/datasets/golden.jsonl --repeats 3 --seed 7 --provider openai
  python -m eval.runners.baseline_b --dataset eval/datasets/golden.jsonl --repeats 3 --seed 7 --provider openai
  ```
